cmake_minimum_required(VERSION 3.16)
project(homework_03 LANGUAGES CXX)

foreach(task IN ITEMS task_1 task_2 task_3 task_4)
    add_executable(${task} ${task}/main.cpp)
    target_compile_features(${task} PRIVATE cxx_std_17)
    set_target_properties(${task} PROPERTIES CXX_EXTENSIONS OFF)
    if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(${task} PRIVATE -Wall -Wextra -Wpedantic)
    endif()
    if(MINGW)
        target_link_options(${task} PRIVATE -static)
    endif()
endforeach()
